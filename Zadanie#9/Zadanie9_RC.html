import numpy as np
import matplotlib.pyplot as plt

x = np.linspace(0, 10, 500)
y1 = np.sin(x)
y2 = np.cos(x)
y3 = np.exp(-x)

plt.plot(x, y1, color='red', linestyle='-', label='sin(x)')
plt.plot(x, y2, color='green', linestyle='--', label='cos(x)')
plt.plot(x, y3, color='blue', linestyle=':', label='exp(-x)')

plt.legend()
plt.xlabel('x')
plt.ylabel('y')
plt.grid()

plt.show()

import numpy as np
import matplotlib.pyplot as plt

n = 100
x = np.random.rand(n)
y = np.random.rand(n)

d = np.sqrt(x**2 + y**2)

colors = np.where(d < 1, 'green', 'red')
sizes = (np.abs(x) + np.abs(y)) * 25

plt.scatter(x, y, c=colors, s=sizes)

plt.axis([-0.05, 1.05, -0.05, 1.05])
plt.xlabel('x')
plt.ylabel('y')
plt.grid()

plt.show()

import numpy as np
import matplotlib.pyplot as plt

def suma_odleglosci(x, y):
    d1 = np.sqrt((x + 1)**2 + y**2)
    d2 = np.sqrt((x - 1)**2 + y**2)
    return d1 + d2

x = np.linspace(-2, 2, 400)
y = np.linspace(-2, 2, 400)

X, Y = np.meshgrid(x, y)

Z = suma_odleglosci(X, Y)

# rysowanie obrazu
plt.imshow(
    Z,
    extent=[-2, 2, -2, 2],
    cmap='bone'
)

plt.colorbar(label='Suma odległości')

plt.xlabel('x')
plt.ylabel('y')
plt.title('Funkcja sumy odległości')

plt.show()
